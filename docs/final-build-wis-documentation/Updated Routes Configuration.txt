import { createBrowserRouter } from 'react-router-dom';

export const router = createBrowserRouter([
  {
    path: "/",
    element: <RootLayout />,
    children: [
      // ... other routes ...
      
      // Knowledge section
      {
        path: "knowledge",
        element: <KnowledgeLayout />,
        children: [
          {
            path: "wis",
            lazy: async () => {
              const { default: Component } = await import('@/pages/knowledge/WISProfessional');
              return { Component };
            },
            loader: async () => {
              // Optional: Preload data
              return {
                title: 'WIS Professional',
                description: 'Workshop Information System'
              };
            }
          }
        ]
      },
      
      // Admin section
      {
        path: "admin",
        element: <AdminLayout />,
        children: [
          {
            path: "wis-management",
            lazy: async () => {
              const { default: Component } = await import('@/pages/admin/WISManagementPage');
              return { Component };
            },
            loader: async ({ request }) => {
              // Check admin access
              const isAdmin = await checkAdminAccess();
              if (!isAdmin) {
                throw new Response("Unauthorized", { status: 403 });
              }
              return { isAdmin };
            }
          }
        ]
      }
    ]
  }
]);
